cmake_minimum_required(VERSION 3.14)
project(cp3 LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(BLAS REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
# Prefer headers beside the chosen library over unrelated system CBLAS headers.
set(CBLAS_HEADER_HINTS)
foreach(blas_library IN LISTS BLAS_LIBRARIES)
    if(IS_ABSOLUTE "${blas_library}")
        get_filename_component(blas_library_dir "${blas_library}" DIRECTORY)
        list(APPEND CBLAS_HEADER_HINTS "${blas_library_dir}/../include"
            "${blas_library_dir}/../include/openblas")
    endif()
endforeach()
find_path(CBLAS_INCLUDE_DIR cblas.h HINTS ${CBLAS_HEADER_HINTS} NO_DEFAULT_PATH)
find_path(CBLAS_INCLUDE_DIR cblas.h PATH_SUFFIXES include include/openblas openblas)

add_library(blas-dependency INTERFACE)
target_link_libraries(blas-dependency INTERFACE ${BLAS_LIBRARIES})
target_compile_features(blas-dependency INTERFACE cxx_std_11)
if(CBLAS_INCLUDE_DIR)
    target_include_directories(blas-dependency INTERFACE ${CBLAS_INCLUDE_DIR})
endif()

add_library(benchmark OBJECT benchmark.cpp)
target_link_libraries(benchmark PUBLIC blas-dependency)

foreach(kind basic vectorized blas openmp)
    add_library(kernel-${kind} OBJECT dgemv-${kind}.cpp)
    target_link_libraries(kernel-${kind} PUBLIC blas-dependency)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(kernel-${kind} PRIVATE -Wall -Wextra -pedantic)
        if(kind STREQUAL "vectorized")
            target_compile_options(kernel-${kind} PRIVATE -O3 -ffast-math)
            if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
                target_compile_options(kernel-${kind} PRIVATE
                    -ftree-vectorize -fopt-info-vec-all=${CMAKE_CURRENT_BINARY_DIR}/report.txt)
            else()
                target_compile_options(kernel-${kind} PRIVATE -Rpass=loop-vectorize
                    -Rpass-missed=loop-vectorize -Rpass-analysis=loop-vectorize)
            endif()
        elseif(NOT kind STREQUAL "blas")
            # Preserve a scalar baseline, including in optimized Release builds.
            target_compile_options(kernel-${kind} PRIVATE -O1)
            if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
                target_compile_options(kernel-${kind} PRIVATE -fno-tree-vectorize)
            else()
                target_compile_options(kernel-${kind} PRIVATE -fno-vectorize -fno-slp-vectorize)
            endif()
        endif()
    endif()
    if(kind STREQUAL "openmp")
        target_link_libraries(kernel-${kind} PUBLIC OpenMP::OpenMP_CXX)
    endif()
    add_executable(benchmark-${kind})
    target_link_libraries(benchmark-${kind} PRIVATE benchmark kernel-${kind})
endforeach()

set(BENCHMARK openmp)
configure_file(job.in job-${BENCHMARK} @ONLY)

include(CTest)
if(BUILD_TESTING)
    foreach(kind basic vectorized openmp)
        add_executable(check-${kind} tests/check_dgemv.cpp)
        target_link_libraries(check-${kind} PRIVATE kernel-${kind})
        if(kind STREQUAL "openmp")
            foreach(threads 1 4 16 64)
                add_test(NAME ${kind}-${threads}-threads COMMAND check-${kind})
                set_tests_properties(${kind}-${threads}-threads PROPERTIES
                    ENVIRONMENT "OMP_NUM_THREADS=${threads};OMP_DYNAMIC=FALSE;OPENBLAS_NUM_THREADS=1;MKL_NUM_THREADS=1")
            endforeach()
        else()
            add_test(NAME ${kind}-correctness COMMAND check-${kind})
            set_tests_properties(${kind}-correctness PROPERTIES
                ENVIRONMENT "OPENBLAS_NUM_THREADS=1;MKL_NUM_THREADS=1")
        endif()
    endforeach()
endif()
